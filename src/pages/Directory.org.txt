import { useEffect, useMemo, useState } from 'react';
import { Link, useSearchParams } from 'react-router-dom';
import {
  Award,
  Globe,
  Loader2,
  MapPin,
  Search,
  Shield,
  Star,
  User,
  X,
} from 'lucide-react';
import { getRaqis } from '@/lib/supabase';
import { useCountries, useCities } from '@/hooks/useCountries';
import { useCountryDetection } from '@/hooks/useCountryDetection';
import type { Raqi } from '@/types';

export default function Directory() {
  const [searchParams, setSearchParams] = useSearchParams();
  const { countryCode, setManualCountry } = useCountryDetection();

  const urlCountry = searchParams.get('country') || '';
  const urlWilaya = searchParams.get('wilaya') || '';

  const [raqis, setRaqis] = useState<Raqi[]>([]);
  const [loading, setLoading] = useState(true);
  const [error, setError] = useState<string | null>(null);
  const [searchQuery, setSearchQuery] = useState('');
  const [selectedCountry, setSelectedCountry] = useState(
    urlCountry || countryCode || 'DZ'
  );
  const [wilayaFilter, setWilayaFilter] = useState(urlWilaya);

  const {
    countries,
    loading: countriesLoading,
    error: countriesError,
  } = useCountries(true);

  const {
    cities,
    loading: citiesLoading,
    error: citiesError,
  } = useCities(selectedCountry);

  useEffect(() => {
    if (urlCountry && urlCountry !== selectedCountry) {
      setSelectedCountry(urlCountry);
      setWilayaFilter('');
    }
  }, [urlCountry, selectedCountry]);

  useEffect(() => {
    if (urlWilaya !== wilayaFilter) {
      setWilayaFilter(urlWilaya);
    }
  }, [urlWilaya, wilayaFilter]);

  useEffect(() => {
    let mounted = true;

    const loadRaqis = async () => {
      setLoading(true);
      setError(null);

      try {
        const data = await getRaqis();
        if (mounted) setRaqis(data);
      } catch (err) {
        console.error('Error loading raqis:', err);
        if (mounted) {
          setRaqis([]);
          setError(
            err instanceof Error
              ? err.message
              : 'تعذر تحميل قائمة الرقاة'
          );
        }
      } finally {
        if (mounted) setLoading(false);
      }
    };

    void loadRaqis();

    return () => {
      mounted = false;
    };
  }, []);

  const updateUrl = (country: string, wilaya: string) => {
    const params = new URLSearchParams(searchParams);

    if (country) params.set('country', country);
    else params.delete('country');

    if (wilaya) params.set('wilaya', wilaya);
    else params.delete('wilaya');

    setSearchParams(params);
  };

  const handleCountryChange = (code: string) => {
    setSelectedCountry(code);
    setWilayaFilter('');
    setManualCountry(code);
    updateUrl(code, '');
  };

  const handleWilayaChange = (value: string) => {
    setWilayaFilter(value);
    updateUrl(selectedCountry, value);
  };

  const clearFilters = () => {
    setSearchQuery('');
    setWilayaFilter('');
    updateUrl(selectedCountry, '');
  };

  const getCountryName = (code?: string) => {
    const country = countries.find((item) => item.code === code);
    return country
      ? `${country.flag_emoji || ''} ${country.name_ar}`.trim()
      : code || 'غير محددة';
  };

  const selectedCountryName = getCountryName(selectedCountry);

  const selectedWilayaName =
    cities.find((city) => city.name_ar === wilayaFilter)?.name_ar ||
    wilayaFilter;

  const raqiGroups = useMemo(() => {
    const emptyGroups = {
      featured: [] as Raqi[],
      verified: [] as Raqi[],
      regular: [] as Raqi[],
    };

    const query = searchQuery.trim().toLowerCase();

    if (!selectedCountry || !wilayaFilter) {
      return emptyGroups;
    }

    const matchesSearch = (raqi: Raqi) => {
      const searchableText = [
        raqi.full_name,
        raqi.speciality,
        raqi.wilaya,
        raqi.address,
        raqi.country_code,
      ]
        .filter(Boolean)
        .join(' ')
        .toLowerCase();

      return !query || searchableText.includes(query);
    };

    const countryRaqis = raqis.filter(
      (raqi) =>
        raqi.country_code === selectedCountry &&
        matchesSearch(raqi)
    );

    // المتميزون يظهرون في جميع ولايات الدولة.
    const featured = countryRaqis.filter(
      (raqi) => Boolean(raqi.featured_badge)
    );

    const featuredIds = new Set(featured.map((raqi) => raqi.id));

    // الموثقون يظهرون في الولاية المختارة فقط.
    // نستبعد المتميزين لمنع تكرار الراقي.
    const verified = countryRaqis.filter(
      (raqi) =>
        !featuredIds.has(raqi.id) &&
        Boolean(raqi.verified_badge) &&
        raqi.wilaya === wilayaFilter
    );

    const usedIds = new Set([
      ...featured.map((raqi) => raqi.id),
      ...verified.map((raqi) => raqi.id),
    ]);

    // بقية الرقاة يظهرون في الولاية المختارة فقط.
    const regular = countryRaqis.filter(
      (raqi) =>
        !usedIds.has(raqi.id) &&
        !raqi.featured_badge &&
        !raqi.verified_badge &&
        raqi.wilaya === wilayaFilter
    );

    return { featured, verified, regular };
  }, [raqis, selectedCountry, wilayaFilter, searchQuery]);

  const totalResults =
    raqiGroups.featured.length +
    raqiGroups.verified.length +
    raqiGroups.regular.length;

  const hasLocation = Boolean(selectedCountry && wilayaFilter);

  return (
    <main className="min-h-screen bg-gray-50 px-4 py-8" dir="rtl">
      <div className="mx-auto max-w-6xl">
        <header className="mb-8 text-center">
          <div className="mb-4 inline-flex items-center gap-2 rounded-full border border-[#bfe6cf] bg-[#ecfdf3] px-4 py-2 text-sm font-bold text-[#166534]">
            <Shield className="h-4 w-4" />
            دليل الرقاة الشرعيين
          </div>

          <h1 className="mb-3 text-3xl font-extrabold text-[#163a2b] md:text-4xl">
            ابحث عن الراقي المناسب
          </h1>

          <p className="text-gray-500">
            اختر الدولة والولاية لعرض الرقاة المتاحين.
          </p>
        </header>

        <section className="mb-8 rounded-3xl border border-gray-200 bg-white p-5 shadow-lg md:p-6">
          <div className="grid grid-cols-1 gap-4 md:grid-cols-3">
            <div>
              <label className="mb-2 block text-sm font-bold text-gray-700">
                <Search className="ml-1 inline-block h-4 w-4" />
                بحث
              </label>

              <div className="relative">
                <Search className="absolute right-3 top-1/2 h-4 w-4 -translate-y-1/2 text-gray-400" />
                <input
                  value={searchQuery}
                  onChange={(event) => setSearchQuery(event.target.value)}
                  placeholder="اسم الراقي أو التخصص..."
                  className="h-12 w-full rounded-xl border border-gray-200 bg-white pl-3 pr-10 text-sm outline-none focus:ring-2 focus:ring-[#1f6f50]"
                />
              </div>
            </div>

            <div>
              <label className="mb-2 block text-sm font-bold text-gray-700">
                <Globe className="ml-1 inline-block h-4 w-4" />
                الدولة
              </label>

              <select
                value={selectedCountry}
                onChange={(event) =>
                  handleCountryChange(event.target.value)
                }
                disabled={countriesLoading}
                className="h-12 w-full rounded-xl border border-gray-200 bg-white px-3 text-sm outline-none focus:ring-2 focus:ring-[#1f6f50]"
              >
                <option value="" disabled>
                  {countriesLoading
                    ? 'جاري تحميل الدول...'
                    : 'اختر الدولة'}
                </option>

                {countries.map((country) => (
                  <option key={country.code} value={country.code}>
                    {country.flag_emoji} {country.name_ar}
                  </option>
                ))}
              </select>
            </div>

            <div>
              <label className="mb-2 block text-sm font-bold text-gray-700">
                <MapPin className="ml-1 inline-block h-4 w-4" />
                الولاية أو المنطقة
              </label>

              <select
                value={wilayaFilter}
                onChange={(event) =>
                  handleWilayaChange(event.target.value)
                }
                disabled={!selectedCountry || citiesLoading}
                className="h-12 w-full rounded-xl border border-gray-200 bg-white px-3 text-sm outline-none focus:ring-2 focus:ring-[#1f6f50]"
              >
                <option value="" disabled>
                  {citiesLoading
                    ? 'جاري تحميل الولايات...'
                    : 'اختر الولاية أو المنطقة'}
                </option>

                {cities.map((city) => (
                  <option key={city.id} value={city.name_ar}>
                    {city.name_ar}
                  </option>
                ))}
              </select>
            </div>
          </div>

          {(countriesError || citiesError) && (
            <p className="mt-4 text-sm text-red-600">
              تعذر تحميل بيانات الدول أو المناطق. حاول تحديث الصفحة.
            </p>
          )}

          {(searchQuery || wilayaFilter) && (
            <button
              type="button"
              onClick={clearFilters}
              className="mt-4 inline-flex items-center gap-2 text-sm font-bold text-red-600 hover:text-red-700"
            >
              <X className="h-4 w-4" />
              مسح الفلاتر
            </button>
          )}
        </section>

        <div className="mb-5 flex items-center justify-between gap-3">
          <h2 className="text-xl font-extrabold text-[#163a2b]">
            {hasLocation ? 'الرقاة المتاحون' : 'اختر الدولة والولاية'}
          </h2>

          <span className="text-sm text-gray-500">
            {hasLocation ? `${totalResults} نتيجة` : ''}
          </span>
        </div>

        {loading && (
          <div className="flex items-center justify-center py-16 text-[#1f6f50]">
            <Loader2 className="h-8 w-8 animate-spin" />
            <span className="mr-3 font-bold">جاري تحميل الرقاة...</span>
          </div>
        )}

        {!loading && error && (
          <div className="rounded-2xl border border-red-200 bg-red-50 p-5 text-center text-red-700">
            {error}
          </div>
        )}

        {!loading && !error && !hasLocation && (
          <div className="rounded-2xl border border-dashed border-gray-300 bg-white p-12 text-center">
            <MapPin className="mx-auto mb-3 h-12 w-12 text-[#1f6f50]" />
            <h3 className="mb-2 text-lg font-bold text-gray-700">
              اختر الدولة والولاية
            </h3>
            <p className="text-sm text-gray-500">
              حدد الدولة أولاً ثم اختر الولاية لعرض الرقاة.
            </p>
          </div>
        )}

        {!loading && !error && hasLocation && totalResults === 0 && (
          <div className="rounded-2xl border border-dashed border-gray-300 bg-white p-12 text-center">
            <User className="mx-auto mb-3 h-12 w-12 text-gray-300" />
            <h3 className="mb-2 text-lg font-bold text-gray-700">
              لا توجد نتائج
            </h3>
            <p className="text-sm text-gray-500">
              لا يوجد رقاة في هذه الولاية حالياً.
            </p>
          </div>
        )}

        {!loading && !error && hasLocation && totalResults > 0 && (
          <div className="space-y-10">
            {raqiGroups.featured.length > 0 && (
              <RaqiSection
                title={`الرقاة المتميزون في ${selectedCountryName}`}
                titleClassName="text-amber-700"
                lineClassName="bg-amber-200"
                raqis={raqiGroups.featured}
                getCountryName={getCountryName}
              />
            )}

            {raqiGroups.verified.length > 0 && (
              <RaqiSection
                title={`الرقاة الموثقون في ${selectedWilayaName}`}
                titleClassName="text-[#166534]"
                lineClassName="bg-green-200"
                raqis={raqiGroups.verified}
                getCountryName={getCountryName}
              />
            )}

            {raqiGroups.regular.length > 0 && (
              <RaqiSection
                title={`بقية الرقاة في ${selectedWilayaName}`}
                titleClassName="text-[#163a2b]"
                lineClassName="bg-gray-200"
                raqis={raqiGroups.regular}
                getCountryName={getCountryName}
              />
            )}
          </div>
        )}
      </div>
    </main>
  );
}

function RaqiSection({
  title,
  titleClassName,
  lineClassName,
  raqis,
  getCountryName,
}: {
  title: string;
  titleClassName: string;
  lineClassName: string;
  raqis: Raqi[];
  getCountryName: (code?: string) => string;
}) {
  return (
    <section>
      <div className="mb-5 flex items-center gap-3">
        <div className={`h-px flex-1 ${lineClassName}`} />
        <h2 className={`text-center text-xl font-extrabold ${titleClassName}`}>
          {title}
        </h2>
        <div className={`h-px flex-1 ${lineClassName}`} />
      </div>

      <div className="grid grid-cols-1 gap-5 sm:grid-cols-2 lg:grid-cols-3">
        {raqis.map((raqi) => (
          <RaqiCard
            key={raqi.id}
            raqi={raqi}
            getCountryName={getCountryName}
          />
        ))}
      </div>
    </section>
  );
}

function RaqiCard({
  raqi,
  getCountryName,
}: {
  raqi: Raqi;
  getCountryName: (code?: string) => string;
}) {
  return (
    <Link to={`/roqat/${raqi.slug}`} className="group">
      <article className="h-full overflow-hidden rounded-2xl border border-gray-200 bg-white shadow-sm transition-all hover:-translate-y-1 hover:shadow-xl">
        <div className="aspect-[4/3] overflow-hidden bg-[#f3f7f4]">
          {raqi.profile_image_url ? (
            <img
              src={raqi.profile_image_url}
              alt={raqi.full_name}
              className="h-full w-full object-cover transition-transform duration-300 group-hover:scale-105"
              loading="lazy"
            />
          ) : (
            <div className="flex h-full w-full items-center justify-center">
              <User className="h-16 w-16 text-[#1f6f50]" />
            </div>
          )}
        </div>

        <div className="p-5">
          <div className="mb-4 flex items-start justify-between gap-3">
            <h3 className="text-lg font-extrabold text-[#163a2b] group-hover:text-[#1f6f50]">
              {raqi.full_name}
            </h3>

            <div className="flex shrink-0 flex-col items-end gap-1">
              {raqi.featured_badge && (
                <span className="inline-flex items-center gap-1 rounded-full border border-amber-200 bg-amber-50 px-2.5 py-1 text-[11px] font-bold text-amber-700">
                  <Star className="h-3 w-3" />
                  متميز
                </span>
              )}

              {raqi.verified_badge && (
                <span className="inline-flex items-center gap-1 rounded-full border border-[#bfe6cf] bg-[#ecfdf3] px-2.5 py-1 text-[11px] font-bold text-[#166534]">
                  <Award className="h-3 w-3" />
                  موثق
                </span>
              )}
            </div>
          </div>

          <div className="space-y-2 text-sm text-gray-600">
            {raqi.speciality && (
              <p className="flex items-center gap-2">
                <Star className="h-4 w-4 text-[#d6b14a]" />
                {raqi.speciality}
              </p>
            )}

            <p className="flex items-center gap-2">
              <Globe className="h-4 w-4 text-[#1f6f50]" />
              {getCountryName(raqi.country_code)}
            </p>

            {raqi.wilaya && (
              <p className="flex items-center gap-2">
                <MapPin className="h-4 w-4 text-[#1f6f50]" />
                {raqi.wilaya}
              </p>
            )}
          </div>

          <div className="mt-5 border-t border-gray-100 pt-4 text-center text-sm font-bold text-[#1f6f50]">
            عرض الملف
          </div>
        </div>
      </article>
    </Link>
  );
}
