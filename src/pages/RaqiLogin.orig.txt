import { useEffect, useState, type FormEvent } from 'react';
import { useNavigate } from 'react-router-dom';
import { Card } from '@/components/ui/card';
import { Button } from '@/components/ui/button';
import { Input } from '@/components/ui/input';
import {
  supabase,
  registerRaqiWithAccount,
  getCurrentRaqiProfile,
} from '@/lib/supabase';
import {
  Shield,
  Lock,
  Mail,
  Loader2,
  LogIn,
  AlertCircle,
  UserPlus,
  ChevronDown,
  ChevronUp,
} from 'lucide-react';

type Wilaya = {
  code: string;
  name_ar: string;
};

export default function RaqiLogin() {
  const navigate = useNavigate();

  const [sessionLoading, setSessionLoading] = useState(true);
  const [sessionError, setSessionError] = useState('');

  const [email, setEmail] = useState('');
  const [password, setPassword] = useState('');
  const [loginError, setLoginError] = useState('');
  const [submittingLogin, setSubmittingLogin] = useState(false);
  const [submittingGoogle, setSubmittingGoogle] = useState(false);

  const [showRegister, setShowRegister] = useState(false);
  const [regEmail, setRegEmail] = useState('');
  const [regPassword, setRegPassword] = useState('');
  const [regFullName, setRegFullName] = useState('');
  const [regWilaya, setRegWilaya] = useState('');
  const [regError, setRegError] = useState('');
  const [submittingReg, setSubmittingReg] = useState(false);
  const [regSuccess, setRegSuccess] = useState(false);

  // ─── ربط راقي قديم بحساب جديد ───
  const [showLink, setShowLink] = useState(false);
  const [linkName, setLinkName] = useState('');
  const [linkWilaya, setLinkWilaya] = useState('');
  const [linkEmail, setLinkEmail] = useState('');
  const [linkPassword, setLinkPassword] = useState('');
  const [linkError, setLinkError] = useState('');
  const [linkLoading, setLinkLoading] = useState(false);
  const [linkSuccess, setLinkSuccess] = useState(false);

  // ─── Wilayas from Supabase ───
  const [wilayas, setWilayas] = useState<Wilaya[]>([]);
  const [loadingWilayas, setLoadingWilayas] = useState(true);

  useEffect(() => {
    const loadWilayas = async () => {
      setLoadingWilayas(true);
      try {
        const { data, error } = await supabase
          .from('wilayas')
          .select('code, name_ar')
          .order('name_ar', { ascending: true });

        if (error) throw error;
        setWilayas(data || []);
      } catch (err) {
        console.error('Load wilayas error:', err);
      } finally {
        setLoadingWilayas(false);
      }
    };

    loadWilayas();
  }, []);

  useEffect(() => {
    let mounted = true;

    const init = async () => {
      try {
        const {
          data: { session },
          error,
        } = await supabase.auth.getSession();

        if (error) throw error;
        if (!mounted) return;

        if (!session) {
          setSessionLoading(false);
          return;
        }

        const profile = await getCurrentRaqiProfile();
        if (!mounted) return;

        if (profile) {
          navigate('/raqi-dashboard', { replace: true });
          return;
        }

        navigate('/raqi-onboarding', { replace: true });
      } catch (error: any) {
        console.error('Session check error:', error);
        if (!mounted) return;
        setSessionError(error?.message || 'تعذر التحقق من الجلسة الحالية.');
      } finally {
        if (mounted) setSessionLoading(false);
      }
    };

    init();

    const {
      data: { subscription },
    } = supabase.auth.onAuthStateChange(async (event, session) => {
      if (!mounted) return;
      if (event === 'SIGNED_OUT' || !session) return;

      try {
        const profile = await getCurrentRaqiProfile();
        if (!mounted) return;

        if (profile) {
          navigate('/raqi-dashboard', { replace: true });
        } else {
          navigate('/raqi-onboarding', { replace: true });
        }
      } catch (error) {
        console.error('Auth state profile check error:', error);
      }
    });

    return () => {
      mounted = false;
      subscription.unsubscribe();
    };
  }, [navigate]);

  const handleLogin = async (e?: FormEvent) => {
    if (e) e.preventDefault();

    if (!email.trim() || !password.trim()) {
      setLoginError('يرجى إدخال البريد الإلكتروني وكلمة المرور');
      return;
    }

    setSubmittingLogin(true);
    setLoginError('');
    setSessionError('');

    try {
      const normalizedEmail = email.trim().toLowerCase();

      const { data, error } = await supabase.auth.signInWithPassword({
        email: normalizedEmail,
        password,
      });

      if (error) {
        if (error.message.toLowerCase().includes('not confirmed')) {
          setLoginError('لم يتم تأكيد البريد الإلكتروني بعد. يرجى التحقق من صندوق الوارد.');
        } else {
          setLoginError('البريد الإلكتروني أو كلمة المرور غير صحيحة.');
        }
        return;
      }

      if (!data.session) {
        setLoginError('تعذر إنشاء الجلسة. يرجى المحاولة مرة أخرى.');
        return;
      }

      const profile = await getCurrentRaqiProfile();

      if (profile) {
        navigate('/raqi-dashboard', { replace: true });
      } else {
        navigate('/raqi-onboarding', { replace: true });
      }
    } catch (err: any) {
      console.error('Login error:', err);
      setLoginError(err?.message || 'حدث خطأ غير متوقع أثناء تسجيل الدخول.');
    } finally {
      setSubmittingLogin(false);
    }
  };

  const handleGoogleLogin = async () => {
    setSubmittingGoogle(true);
    setLoginError('');
    setSessionError('');

    try {
      const { error } = await supabase.auth.signInWithOAuth({
        provider: 'google',
        options: {
          redirectTo: `${window.location.origin}/auth/callback`,
          queryParams: {
            prompt: 'select_account',
          },
        },
      });

      if (error) {
        setLoginError(error.message || 'تعذر المتابعة باستخدام Google.');
      }
    } catch (err: any) {
      console.error('Google login error:', err);
      setLoginError(err?.message || 'حدث خطأ أثناء المتابعة باستخدام Google.');
    } finally {
      setSubmittingGoogle(false);
    }
  };

  const handleQuickRegister = async (e: FormEvent) => {
    e.preventDefault();
    setRegError('');
    setRegSuccess(false);

    if (!regFullName.trim()) {
      setRegError('الاسم الكامل مطلوب.');
      return;
    }

    if (!regEmail.trim() || !/^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(regEmail.trim())) {
      setRegError('يرجى إدخال بريد إلكتروني صحيح.');
      return;
    }

    if (!regPassword.trim() || regPassword.length < 6) {
      setRegError('كلمة المرور يجب أن تكون 6 أحرف على الأقل.');
      return;
    }

    if (!regWilaya) {
      setRegError('يرجى اختيار الولاية.');
      return;
    }

    setSubmittingReg(true);

    try {
      const normalizedEmail = regEmail.trim().toLowerCase();

      const result = await registerRaqiWithAccount({
        email: normalizedEmail,
        password: regPassword,
        full_name: regFullName.trim(),
        wilaya: regWilaya,
      });

      if (result?.needsEmailConfirmation) {
        setRegSuccess(true);
        setRegError('');
        return;
      }

      setRegSuccess(true);

      const { data, error: loginErr } = await supabase.auth.signInWithPassword({
        email: normalizedEmail,
        password: regPassword,
      });

      if (loginErr) {
        setRegError('تم إنشاء الحساب، لكن تعذر تسجيل الدخول تلقائيًا.');
        return;
      }

      if (!data.session) {
        setRegError('تم إنشاء الحساب، لكن تعذر إنشاء الجلسة.');
        return;
      }

      const profile = await getCurrentRaqiProfile();

      if (profile) {
        navigate('/raqi-dashboard', { replace: true });
      } else {
        navigate('/raqi-onboarding', { replace: true });
      }
    } catch (err: any) {
      console.error('Register error:', err);
      setRegError(err?.message || 'فشل إنشاء الحساب.');
    } finally {
      setSubmittingReg(false);
    }
  };

  const handleLinkExistingRaqi = async (e: FormEvent) => {
  e.preventDefault();
  setLinkError('');
  setLinkSuccess(false);

  const name = linkName.trim();
  const wilaya = linkWilaya.trim();
  const email = linkEmail.trim().toLowerCase();
  const password = linkPassword;

  // 🔍 للتشخيص
  console.log('🔍 Link attempt:', { name, wilaya, email });

  if (!name || !wilaya) {
    setLinkError('الاسم الكامل والولاية مطلوبان.');
    return;
  }
  if (!email || !/^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(email)) {
    setLinkError('يرجى إدخال بريد إلكتروني صحيح.');
    return;
  }
  if (!password || password.length < 6) {
    setLinkError('كلمة المرور يجب أن تكون 6 أحرف على الأقل.');
    return;
  }

  setLinkLoading(true);

  try {
    // 1) البحث عن الراقي في جدول raqis
    const { data: raqis, error: raqisError } = await supabase
      .from('raqis')
      .select('*')
      .ilike('full_name', name)
      .eq('wilaya', wilaya);

    // 🔍 للتشخيص
    console.log('🔍 Query result:', { raqis, raqisError });

    if (raqisError) throw raqisError;

    if (!raqis || raqis.length === 0) {
      setLinkError('لم يتم العثور على راقي بهذا الاسم والولاية.');
      return;
    }
    // ... بقية الكود
      if (raqis.length > 1) {
        setLinkError('تم العثور على أكثر من راقي بهذا الاسم والولاية. يرجى التواصل مع الإدارة.');
        return;
      }

      const raqi = raqis[0];

      if (raqi.has_auth_account && raqi.user_id) {
        setLinkError('هذا الراقي لديه حساب دخول بالفعل. يرجى استخدام تسجيل الدخول العادي.');
        return;
      }

      // 2) إنشاء مستخدم Auth جديد
      const { data: signUpData, error: signUpError } = await supabase.auth.signUp({
        email,
        password,
        options: {
          data: {
            full_name: raqi.full_name,
          },
        },
      });

      if (signUpError || !signUpData.user) {
        throw signUpError || new Error('تعذر إنشاء حساب الدخول');
      }

      const user = signUpData.user;

      // 3) ربط صف raqis بالمستخدم الجديد وتحديث الإيميل
      const { error: updateError } = await supabase
        .from('raqis')
        .update({
          user_id: user.id,
          email,
          has_auth_account: true,
        })
        .eq('id', raqi.id);

      if (updateError) throw updateError;

      // 4) محاولة تسجيل الدخول مباشرة
      const { error: loginError } = await supabase.auth.signInWithPassword({
        email,
        password,
      });

      if (loginError) {
        setLinkSuccess(true);
        setLinkError(
          'تم ربط حسابك بنجاح، لكن يجب تأكيد البريد الإلكتروني قبل تسجيل الدخول.',
        );
        return;
      }

      // 5) التحقق من الملف الشخصي والتوجيه
      const profile = await getCurrentRaqiProfile();

      if (profile) {
        navigate('/raqi-dashboard', { replace: true });
      } else {
        navigate('/raqi-onboarding', { replace: true });
      }
    } catch (err: any) {
      console.error('Link existing raqi error:', err);
      setLinkError(err?.message || 'حدث خطأ أثناء ربط الراقي بالحساب.');
    } finally {
      setLinkLoading(false);
    }
  };

  if (sessionLoading) {
    return (
      <div className="min-h-screen bg-[#f6f7f4] flex items-center justify-center px-4">
        <div className="flex items-center gap-3 text-gray-700 font-bold">
          <Loader2 className="w-6 h-6 animate-spin text-[#1f6f50]" />
          جاري التحقق...
        </div>
      </div>
    );
  }

  return (
    <div className="min-h-screen relative overflow-hidden bg-[#f6f7f4] px-4 py-10">
      <div className="absolute inset-0 pointer-events-none">
        <div className="absolute top-0 left-0 w-[420px] h-[420px] bg-[#1f6f50]/[0.06] rounded-full blur-3xl" />
        <div className="absolute bottom-0 right-0 w-[380px] h-[380px] bg-[#d6b14a]/[0.08] rounded-full blur-3xl" />
      </div>

      <div className="relative max-w-5xl mx-auto grid lg:grid-cols-2 gap-6">
        {/* بطاقة تسجيل الدخول */}
        <Card className="rounded-[32px] border border-white/70 bg-white/95 backdrop-blur shadow-[0_20px_60px_rgba(16,24,40,0.08)] p-6 md:p-8">
          <div className="text-center mb-8">
            <div className="w-18 h-18 rounded-[24px] bg-gradient-to-br from-[#1f6f50]/15 to-[#d6b14a]/15 border border-[#1f6f50]/10 flex items-center justify-center mx-auto mb-5">
              <Shield className="w-8 h-8 text-[#1f6f50]" />
            </div>

            <span className="inline-flex items-center rounded-full border border-[#1f6f50]/10 bg-[#f5faf7] px-4 py-1.5 text-xs font-bold text-[#1f6f50] mb-4">
              دخول آمن وسريع
            </span>

            <h1 className="text-3xl md:text-4xl font-extrabold text-gray-900 leading-tight">
              تسجيل دخول الراقي
            </h1>

            <p className="text-gray-500 mt-3 leading-relaxed max-w-md mx-auto">
              ادخل إلى حسابك لمتابعة إدارة ملفك والوصول إلى لوحة الراقي.
            </p>
          </div>

          {(sessionError || loginError) && (
            <div className="mb-4 rounded-2xl border border-red-200 bg-red-50 px-4 py-3 text-sm text-red-700 font-semibold flex items-start gap-2">
              <AlertCircle className="w-5 h-5 shrink-0 mt-0.5" />
              <span>{sessionError || loginError}</span>
            </div>
          )}

          <div className="space-y-3">
            <Button
              type="button"
              onClick={handleGoogleLogin}
              disabled={submittingGoogle}
              variant="outline"
              className="w-full h-14 rounded-2xl border border-black/10 bg-[#1f1d17] hover:bg-[#171510] text-white font-extrabold shadow-sm transition-all"
            >
              {submittingGoogle ? (
                <>
                  <Loader2 className="w-4 h-4 ml-2 animate-spin" />
                  جاري التحويل إلى Google...
                </>
              ) : (
                <>
                  <span className="flex items-center justify-center w-7 h-7 rounded-full bg-white ml-3 shrink-0">
                    <svg className="w-4 h-4" viewBox="0 0 24 24" aria-hidden="true">
                      <path
                        fill="#EA4335"
                        d="M12 10.2v3.9h5.5c-.2 1.3-1.5 3.9-5.5 3.9-3.3 0-6-2.7-6-6s2.7-6 6-6c1.9 0 3.2.8 3.9 1.5l2.7-2.6C16.9 3.3 14.7 2.4 12 2.4 6.7 2.4 2.4 6.7 2.4 12S6.7 21.6 12 21.6c6.9 0 9.1-4.8 9.1-7.3 0-.5-.1-.9-.1-1.3H12z"
                      />
                    </svg>
                  </span>
                  المتابعة باستخدام Google
                </>
              )}
            </Button>

            <p className="text-center text-xs text-gray-400 leading-relaxed px-2">
              بالمتابعة، فإنك توافق على سياسة الخصوصية وشروط استخدام المنصة.
            </p>
          </div>

          <div className="my-6 flex items-center gap-3">
            <div className="h-px flex-1 bg-gray-200" />
            <span className="text-sm text-gray-400 font-semibold">أو سجّل بالبريد الإلكتروني</span>
            <div className="h-px flex-1 bg-gray-200" />
          </div>

          <form onSubmit={handleLogin} className="space-y-4">
            <div className="relative">
              <Mail className="w-5 h-5 text-gray-400 absolute right-4 top-1/2 -translate-y-1/2" />
              <Input
                type="email"
                value={email}
                onChange={(e) => {
                  setEmail(e.target.value);
                  setLoginError('');
                  setSessionError('');
                }}
                placeholder="أدخل بريدك الإلكتروني"
                className="h-14 pr-11 rounded-2xl border-gray-200 bg-[#fcfcfb] focus-visible:ring-[#1f6f50]"
                dir="ltr"
              />
            </div>

            <div className="relative">
              <Lock className="w-5 h-5 text-gray-400 absolute right-4 top-1/2 -translate-y-1/2" />
              <Input
                type="password"
                value={password}
                onChange={(e) => {
                  setPassword(e.target.value);
                  setLoginError('');
                  setSessionError('');
                }}
                placeholder="أدخل كلمة المرور"
                className="h-14 pr-11 rounded-2xl border-gray-200 bg-[#fcfcfb] focus-visible:ring-[#1f6f50]"
                dir="ltr"
              />
            </div>

            <Button
              type="submit"
              disabled={submittingLogin}
              className="w-full h-14 rounded-2xl bg-[#1f6f50] hover:bg-[#18593f] text-white font-extrabold shadow-[0_12px_30px_rgba(31,111,80,0.22)] transition-all"
            >
              {submittingLogin ? (
                <>
                  <Loader2 className="w-4 h-4 ml-2 animate-spin" />
                  جاري تسجيل الدخول...
                </>
              ) : (
                <>
                  <LogIn className="w-4 h-4 ml-2" />
                  المتابعة بالبريد الإلكتروني
                </>
              )}
            </Button>
          </form>
        </Card>

        {/* البطاقة الخاصة بإنشاء حساب جديد + ربط راقي قديم */}
        <div className="space-y-6">
          {/* إنشاء حساب جديد */}
          <Card className="rounded-[32px] border border-white/70 bg-white/95 backdrop-blur shadow-[0_20px_60px_rgba(16,24,40,0.08)] p-6 md:p-8">
            <button
              type="button"
              onClick={() => setShowRegister((v) => !v)}
              className="w-full flex items-center justify-between text-right"
            >
              <div>
                <span className="inline-flex items-center rounded-full border border-[#d6b14a]/20 bg-[#fff9ec] px-4 py-1.5 text-xs font-bold text-[#9d7d22] mb-4">
                  إنشاء حساب جديد
                </span>

                <h2 className="text-2xl md:text-3xl font-extrabold text-gray-900">
                  افتح حسابك بسرعة
                </h2>

                <p className="text-gray-500 mt-3 leading-relaxed">
                  أنشئ حسابك أولًا، ثم أكمل بيانات ملف الراقي بعد الدخول.
                </p>
              </div>

              <div className="w-11 h-11 rounded-2xl bg-[#1f6f50]/10 flex items-center justify-center text-[#1f6f50] shrink-0">
                {showRegister ? <ChevronUp className="w-5 h-5" /> : <ChevronDown className="w-5 h-5" />}
              </div>
            </button>

            {showRegister && (
              <form onSubmit={handleQuickRegister} className="space-y-4 mt-7">
                <Input
                  value={regFullName}
                  onChange={(e) => {
                    setRegFullName(e.target.value);
                    setRegError('');
                  }}
                  placeholder="الاسم الكامل"
                  className="h-14 rounded-2xl border-gray-200 bg-[#fcfcfb]"
                />

                <Input
                  type="email"
                  value={regEmail}
                  onChange={(e) => {
                    setRegEmail(e.target.value);
                    setRegError('');
                  }}
                  placeholder="البريد الإلكتروني"
                  className="h-14 rounded-2xl border-gray-200 bg-[#fcfcfb]"
                  dir="ltr"
                />

                <Input
                  type="password"
                  value={regPassword}
                  onChange={(e) => {
                    setRegPassword(e.target.value);
                    setRegError('');
                  }}
                  placeholder="كلمة المرور"
                  className="h-14 rounded-2xl border-gray-200 bg-[#fcfcfb]"
                  dir="ltr"
                />

                <select
                  value={regWilaya}
                  onChange={(e) => {
                    setRegWilaya(e.target.value);
                    setRegError('');
                  }}
                  disabled={loadingWilayas}
                  className="w-full h-14 px-4 border border-gray-200 rounded-2xl text-base bg-[#fcfcfb] text-gray-800 focus:outline-none focus:ring-2 focus:ring-[#1f6f50] focus:border-transparent disabled:opacity-70 disabled:cursor-wait"
                >
                  <option value="">
                    {loadingWilayas ? 'جاري تحميل الولايات...' : 'اختر الولاية'}
                  </option>
                  {wilayas.map((w) => (
                    <option key={w.code} value={w.name_ar}>
                      {w.name_ar}
                    </option>
                  ))}
                </select>

                {regError && (
                  <div className="rounded-2xl border border-red-200 bg-red-50 px-4 py-3 text-sm text-red-700 font-semibold flex items-start gap-2">
                    <AlertCircle className="w-5 h-5 shrink-0 mt-0.5" />
                    <span>{regError}</span>
                  </div>
                )}

                {regSuccess && (
                  <div className="rounded-2xl border border-green-200 bg-green-50 px-4 py-3 text-sm text-green-700 font-semibold leading-relaxed">
                    تم إنشاء الحساب بنجاح. يمكنك الآن تسجيل الدخول مباشرة
                  </div>
                )}

                <Button
                  type="submit"
                  disabled={submittingReg}
                  className="w-full h-14 rounded-2xl bg-[#d6b14a] hover:bg-[#c5a13f] text-white font-extrabold shadow-[0_12px_30px_rgba(214,177,74,0.24)] transition-all"
                >
                  {submittingReg ? (
                    <>
                      <Loader2 className="w-4 h-4 ml-2 animate-spin" />
                      جاري إنشاء الحساب...
                    </>
                  ) : (
                    <>
                      <UserPlus className="w-4 h-4 ml-2" />
                      إنشاء الحساب
                    </>
                  )}
                </Button>
              </form>
            )}
          </Card>

          {/* ربط راقي قديم بحساب جديد */}
          <Card className="rounded-[32px] border border-white/70 bg-white/95 backdrop-blur shadow-[0_20px_60px_rgba(16,24,40,0.08)] p-6 md:p-8">
            <button
              type="button"
              onClick={() => setShowLink((v) => !v)}
              className="w-full flex items-center justify-between text-right"
            >
              <div>
                <span className="inline-flex items-center rounded-full border border-[#1f6f50]/20 bg-[#f5faf7] px-4 py-1.5 text-xs font-bold text-[#1f6f50] mb-4">
                  ربط ملف قديم
                </span>

                <h2 className="text-2xl md:text-3xl font-extrabold text-gray-900">
                  ربط راقي موجود مسبقًا بحساب جديد
                </h2>

                <p className="text-gray-500 mt-3 leading-relaxed">
                  إذا كنت راقياً مضافاً مسبقاً في الدليل بدون حساب دخول، يمكنك إنشاء حساب وربطه بملفك.
                </p>
              </div>

              <div className="w-11 h-11 rounded-2xl bg-[#1f6f50]/10 flex items-center justify-center text-[#1f6f50] shrink-0">
                {showLink ? <ChevronUp className="w-5 h-5" /> : <ChevronDown className="w-5 h-5" />}
              </div>
            </button>

            {showLink && (
              <form onSubmit={handleLinkExistingRaqi} className="space-y-4 mt-7">
                <Input
                  value={linkName}
                  onChange={(e) => {
                    setLinkName(e.target.value);
                    setLinkError('');
                  }}
                  placeholder="الاسم الكامل كما في الدليل"
                  className="h-14 rounded-2xl border-gray-200 bg-[#fcfcfb]"
                />

                <select
                  value={linkWilaya}
                  onChange={(e) => {
                    setLinkWilaya(e.target.value);
                    setLinkError('');
                  }}
                  disabled={loadingWilayas}
                  className="w-full h-14 px-4 border border-gray-200 rounded-2xl text-base bg-[#fcfcfb] text-gray-800 focus:outline-none focus:ring-2 focus:ring-[#1f6f50] focus:border-transparent disabled:opacity-70 disabled:cursor-wait"
                >
                  <option value="">
                    {loadingWilayas ? 'جاري تحميل الولايات...' : 'اختر الولاية'}
                  </option>
                  {wilayas.map((w) => (
                    <option key={w.code} value={w.name_ar}>
                      {w.name_ar}
                    </option>
                  ))}
                </select>

                <Input
                  type="email"
                  value={linkEmail}
                  onChange={(e) => {
                    setLinkEmail(e.target.value);
                    setLinkError('');
                  }}
                  placeholder="البريد الإلكتروني الجديد"
                  className="h-14 rounded-2xl border-gray-200 bg-[#fcfcfb]"
                  dir="ltr"
                />

                <Input
                  type="password"
                  value={linkPassword}
                  onChange={(e) => {
                    setLinkPassword(e.target.value);
                    setLinkError('');
                  }}
                  placeholder="كلمة المرور (6 أحرف على الأقل)"
                  className="h-14 rounded-2xl border-gray-200 bg-[#fcfcfb]"
                  dir="ltr"
                />

                {linkError && (
                  <div className="rounded-2xl border border-red-200 bg-red-50 px-4 py-3 text-sm text-red-700 font-semibold flex items-start gap-2">
                    <AlertCircle className="w-5 h-5 shrink-0 mt-0.5" />
                    <span>{linkError}</span>
                  </div>
                )}

                {linkSuccess && (
                  <div className="rounded-2xl border border-green-200 bg-green-50 px-4 py-3 text-sm text-green-700 font-semibold leading-relaxed">
                    تم ربط حسابك بنجاح. يمكنك الآن تسجيل الدخول مباشرة.
                  </div>
                )}

                <Button
                  type="submit"
                  disabled={linkLoading}
                  className="w-full h-14 rounded-2xl bg-[#1f6f50] hover:bg-[#18593f] text-white font-extrabold shadow-[0_12px_30px_rgba(31,111,80,0.22)] transition-all"
                >
                  {linkLoading ? (
                    <>
                      <Loader2 className="w-4 h-4 ml-2 animate-spin" />
                      جاري ربط الحساب...
                    </>
                  ) : (
                    <>
                      <UserPlus className="w-4 h-4 ml-2" />
                      إنشاء حساب وربط الملف
                    </>
                  )}
                </Button>
              </form>
            )}
          </Card>
        </div>
      </div>
    </div>
  );
}
